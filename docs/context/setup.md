# Cài đặt .NET 8 SDK
winget install Microsoft.DotNet.SDK.8

# Cài đặt Visual Studio 2022 Community kèm workload .NET desktop
winget install Microsoft.VisualStudio.2022.Community --override "--add Microsoft.VisualStudio.Workload.ManagedDesktop"

# Cài đặt các công cụ bổ trợ
winget install Git.Git
winget install GitHub.GitLFS
winget install Gyan.FFmpeg
winget install 7zip.7zip
winget install Microsoft.Sysinternals
